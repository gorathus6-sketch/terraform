# grep for patterns
grep -i <string or pattern value> <filename>    # can be partial filename using * as a wildcard such as *.rpt or *OCT*

grep -r <string> <filename>    # this will search recursively through sub dirs

grep -iE "error|fail|exception"

# trailers and headers
tail -f <filename>    # to see an active log or actively written file increment in real time
tail -f <logfile> | grep --line-buffered ERROR    # real time error capture
tail -<number> <filename>    # Shows the speficied number of rows towards EOF, defaults to -10
head -<number> <filename>    # shows the specified number of rows towards BOF, defaults to -10

# changing ownership and permissions

chmod XXX <filename>    # XXX can denote 1-7, 1 = execute, 2 = write, 4 = read, 5 = read + exec, 6 = read+write, 7 = read+write+exec. First octet (X--) symbolizes the owner, second octet (-X-) the owner's group, third octet (--X) symbolizes user not associated with either of those, or "others"

chown <user>:<group> <filename>    # assigns a user and associated group to the file

# add sticky bit

chmod +t

# add setuid

chmod u+s

# set gid

chmod g+s

# Also there are UMASK, which is an inversion of sorts for the set permissions, ie 022 = 755


# finding files

find <directory> -name <filename>    # E.g. find /ach -name ach_mmddyyyy.ach
# also do find /ach -iname ach_mmddyyyy.ach 2> /dev/null to make the search non-case sensitive, and not return error messages


# awk
awk '{print $x}' <filename>    # x denotes the column number of the target file
awk '/error/' <filename>    # this will print lines which contain strings matching the pattern "error" 

awk '{sum += $x} END {print sum}' <filename>    # sum values in a column, denoted by $x

# e.g.
awk '{sum += $1} END {print sum}' test_csv.csv
12

# Use a colon as a separator
# (e.g., for /etc/passwd):
awk -F: '{print $1}' /etc/passwd
# returns list of accounts, user or system owned

awk '{print $1}' | sort | uniq -c | sort -nr    # frequency analysis


# tee command
<command> | sudo tee -a <output_file>    # duplicates output of command to a file in real time

# diff
diff -y <filename1> <filename2>    # compares files side by side
# exit status: 0 = files are identical, 1 = differences in the files, 2 = errors encountered

# word count
wc -l    # counts number of records, important if an ACH or statement file has an expected number of records, but records are missing


# System health

# processes

top
htop    # these will list processes in terms of CPU time first, memory second, great for identifying CPU hogs

free -m     # this will show you free (and available) RAM in MB, as well as SWAP space memory

ps -ef    # this will return the list of PIDs (processes taking CPU time)

ps -fp <pid>    # see command line assosiated with the process

ps aux --sort=-rss | head    # shows top memory hogs

lsof -p <pid>    # see open files

strace -p <pid>    # see system calls (if permissed or able)

nice     #or renice    # reassigns process priority

# if there is a zombie process (or if a process is no longer required to run:
kill -9 <pid>

pmap <pid>    # memory map


# disk space

df -h    # shows free space in available file systems

du -sh    # shows how much space files/folders are taking in any given directory

# Compress older/archived files to save space

find /path/to/directory -type f -mtime +7 -exec gzip {} +

# or in current directory:

find . -type f -mtime +7 -exec gzip {} +    # this will compress files at least 7 days old


# diagnose disk IO bottlenecks

iostat -xz 1    # I/O bottlenecks

vmstat 1

iotop    # per process IO


# diagnose network issues

ss -tulpn    # listening ports

netstat -anp    # print network connections, routing tables, interface statistics, masquerade connections, and multicast memberships

lsof -i :<port_num>    # process using specified port number

curl -v    # endpoint debugging

dig/nslookup    # DNS

# -- check for table space alerts

WITH TARGET_TABLES AS (
    SELECT 'BANK' AS TBL_NAME FROM DUAL
    UNION ALL
    SELECT 'MESSAGE_FEED' FROM DUAL
    UNION ALL
    SELECT 'USER' FROM DUAL
    UNION ALL
    SELECT 'SHORT_CODE' FROM DUAL
)
SELECT
    T.TBL_NAME,
    A.ALERT_TYPE,
    A.ALERT_DATE,
    A.SEVERITY,
    TS.TABLESPACE_NAME,
FROM TARGET_TABLES T
LEFT JOIN ALERTS A
    ON T.TBL_NAME = A.OBJECT_NAME
LEFT JOIN USER_TABLESPACES TS
    ON TS.TABLESPACE_NAME = (
        SELECT TABLESPACE_NAME
        FROM USER_TABLES
        WHERE TABLE_NAME = T.TBL_NAME
    )
ORDER BY A.ALERT_DATE DESC;

    lo.session_id,
    s.serial#,
    s.username,
    s.osuser,
    s.machine,
    o.object_name,
    lo.locked_mode
FROM v$locked_object lo
JOIN dba_objects o ON lo.object_id = o.object_id
JOIN v$session s ON lo.session_id = s.sid
WHERE o.object_name = 'BANK'

# -- check for long running queries / sessions

SET PAGESIZE 0
SET LINESIZE 200
SET FEEDBACK OFF
SET HEADING OFF
SET VERIFY OFF
SET ECHO OFF
SET TERMOUT OFF

-- Threshold in seconds (modify as needed)
DEFINE THRESHOLD = 300;

-- Query for long-running active sessions
SELECT 'LONG_RUNNING: SID=' || s.sid ||
       ' SERIAL=' || s.serial# ||
       ' USER=' || s.username ||
       ' SQL_ID=' || s.sql_id ||
       ' SECONDS=' || s.last_call_et
FROM v$session s
WHERE s.status = 'ACTIVE'
  AND s.username IS NOT NULL
  AND s.last_call_et > &THRESHOLD;

# JOB CONTROL LANGUAGE

# Most prod support roles will be using batch processing schedulers such as cron, autosys, zena, airflow, control-m, dollar universe, etc.
# Let's start with autosys

insert_job: <job_name>
job_type: c
command: <path to the target batch script>
machine: <target_hostname>
condition: success(parent_job_name)

# other job types:
# BOX, a container which contains related batch jobs
# FW, file watcher, checks for file required for command job to run and ingest the file

# Check status of a batch job
autorep -j <jobname>

# autosys events
FORCE_STARTJOB    # forces a job to run regardless if condition is met or not
JOB_ON_HOLD    # suspends a job, and this will suspend all dependencies as well
JOB_OFF_HOLD    # releases a job from suspension
JOB_ON_ICE    # suspends a job without suspending the dependent jobs, effectively skipping it
JOB_OFF_ICE    # will unsuspend (or unskip) the job
CHANGE_STATUS    # manually override job status to success or failure

# Alerts
CHASE    # indicates that the scheduler cannot verify the status of a running or starting job via the remote agen
MAX_RUN_ALARM    # indicates job has run past it's expected run time


# Job Conditions
s(parent_job)    # success
f(parent_job)    # failure
d(parent_job)    # done
n(parent_job)    # not running

# grep for patterns
grep -i <string or pattern value> <filename>    # can be partial filename using * as a wildcard such as *.rpt or *OCT*

grep -r <string> <filename>    # this will search recursively through sub dirs

grep -iE "error|fail|exception"    # multi pattern search

grep -A3 -B3 “ERROR” logfile    # shows context around errors, with ERROR being highlighted

# trailers and headers
tail -f <filename>    # to see an active log or actively written file increment in real time
tail -f <logfile> | grep --line-buffered ERROR    # real time error capture
tail -<number> <filename>    # Shows the speficied number of rows towards EOF, defaults to -10
head -<number> <filename>    # shows the specified number of rows towards BOF, defaults to -10

# note: checking header or trailer at the BOF or EOF is important to ensure the file's date matches with the
# intended business day (in the header) and files such as ACH or other transactions files should contain
# both the transaction amount and record amount (in the trailer).


# changing ownership and permissions

chmod XXX <filename>    # X's in XXX can denote 1-7, 1 = execute, 2 = write, 4 = read, 5 = read + exec, 6 = read+write, 7 = read+write+exec. First octet (X--) symbolizes the owner, second octet (-X-) the owner's group,
# third octet (--X) symbolizes user not associated with either of those, or "others"

chown <user>:<group> <filename>    # assigns a user and associated group to the file

# add sticky bit, this will prevent users from deleting or renaming files they don't own

chmod +t filename

# add setuid, a special file permission that allows a user to run an executable with the permissions of the file's owner instead of their own

chmod u+s filename

# set gid (group id), when the file is executed, the process runs with the privileges of the group that owns the file, rather than the group of the user who started it.

chmod g+s filename

# Also there are UMASK, which is an inversion of sorts for the set permissions, ie 022 = 755. Sets rules for future files which don't exist yet.

# -- Atomicity, Consistency, Isolation, Durability (ACID)

# -- Atomicity (All or Nothing): Every step in a transaction is treated as a single, indivisible unit. Either the entire transaction succeeds and is saved (committed), or if any part fails, everything is undone
(rolled back) to the previous state.
# --Example: If you transfer $100 from Account A to Account B, atomicity ensures you don't end up in a situation where the money leaves A but never reaches B.

# -- Consistency (Valid States Only): A transaction must move the database from one valid state to another, following all predefined rules and constraints, such as unique keys or foreign keys. If a transaction
would violate these rules, it is canceled.

# -- Isolation (Independent Execution): Multiple transactions happening at the same time should not interfere with each other. Each transaction should run as if it were the only one happening in the system.
# -- Example: If two people try to book the last available seat on a flight simultaneously, isolation ensures only one succeeds and the other sees the seat as already taken.

# -- Durability (Permanent Changes): Once a transaction is successfully committed, the changes are permanent and cannot be lost, even if the system crashes or loses power immediately after. This is typically
achieved by saving the data to non-volatile storage like a disk.


# -- Transaction Isolation Levels

# -- Read Committed: default

# -- Repeatable Read: prevent non-repeatable reads    # --In SQL, a non-repeatable read is a concurrency problem that occurs when a transaction reads the same row twice but finds that the data has changed between those reads. This happens because another transaction modified and committed changes to that specific row after the first read but before the second.
 
# -- serializable - strict, prevent phantom reads    # -- A phantom read in SQL occurs when a transaction runs the same query twice and gets a different set of rows each time because another transaction inserted or deleted data in between

# -- check for table space alerts

WITH TARGET_TABLES AS (                                               -- start common table expression CTE, ‘Target_Tables’
    SELECT 'BANK' AS TBL_NAME FROM DUAL                               -- add first row, table name ‘Bank”
    UNION ALL                                                         -- UNION ALL keeps duplicates, faster than Union
    SELECT 'MESSAGE_FEED' FROM DUAL                                   -- add second row: Message_Feed
    UNION ALL
    SELECT 'USER' FROM DUAL                                           -- add third row: User
    UNION ALL
    SELECT 'SHORT_CODE' FROM DUAL                                     -- add fourth row: Short_Code
)
SELECT
    T.TBL_NAME,                                                       -- return table name from the CTE
    A.ALERT_TYPE,                                                     -- alert type from Alerts table, can be null if no match
    A.ALERT_DATE,                                                     -- alert date from Alerts table
    A.SEVERITY,                                                       -- severity of the alert
    TS.TABLESPACE_NAME,                                               -- tablespace name from User_Tablespaces
FROM TARGET_TABLES T                                                  -- Use the common table expression as the driving table
LEFT JOIN ALERTS A                                                    -- Left Join so all Target_Tables rows appear, even if
                                                                      -- no alerts
    ON T.TBL_NAME = A.OBJECT_NAME                                     -- match alerts where Object_Name equals tablename 
LEFT JOIN USER_TABLESPACES TS                                         -- Left Join to get table space info
    ON TS.TABLESPACE_NAME = (                                         -- match on tablespacename returned by subquery
        SELECT TABLESPACE_NAME                                        -- subquery: find the tablespace for this table
        FROM USER_TABLES                                              -- User_Tables contains metadata about user owned tables
        WHERE TABLE_NAME = T.TBL_NAME                                 -- Match the table name from the CTE
    )
ORDER BY A.ALERT_DATE DESC;                                           -- Sort results by the most recent alert first

# -- NOTE: This query starts by defining a Common Table Expression called TARGET_TABLES.
Inside the CTE, essentially hard‑coding a small list of table names — BANK, MESSAGE_FEED, USER, and SHORT_CODE. Use UNION ALL because one doesn’t need deduplication and it’s more efficient.

# -- In the main query, select from that CTE and treat it as my driving table.

# -- From there, left‑join into an ALERTS table, matching each table name to any alert records where the alert’s OBJECT_NAME equals the table name. Use a LEFT JOIN so that even if a table has no alerts, it still
# -- appears in the output.

# -- Next, left‑join into USER_TABLESPACES, but the join condition uses a subquery.
# -- That subquery looks up the tablespace for each table by querying USER_TABLES and matching on the table name. So effectively, one is resolving each table’s tablespace dynamically.

# -- Finally, order the results by alert date in descending order so the most recent alerts appear first."


# -- checking for data base locks

SELECT
    lo.session_id,        -- IDs the unique ID of the session holding the lock 
    s.serial#,            -- serial num of the session used with ID to uniquely ID or kill a session
    s.username,           -- database schema name associated with the session
    s.osuser,             -- O/S username of person or process connected
    s.machine,            -- name of computer or server where connection originated
    o.object_name,        -- name of the table or index that is being locked
    lo.locked_mode        -- numeric code representing type of lock (row share, exclusive, etc)
FROM v$locked_object lo   -- pulls from dynamic performance view, lists every active lock in
                          -- the system; labeled ‘lo’ for short
JOIN dba_objects o ON lo.object_id = o.object_id
  -- connects the lock info to the master list of objects to see the actual name (Bank) rather
  -- than just a numeric ID
JOIN v$session s ON lo.session_id = s.sid
  -- connects the lock info to the session table to see who actually logged in and from which
  -- machine
WHERE o.object_name = 'BANK'
  -- limits the output to only show locks affecting table named ‘Bank’

-- Threshold in seconds (modify as needed)
DEFINE THRESHOLD = 300;                           -- creates reusable variable so you don’t have to hunt thru the
                                                  -- code to change it

-- Query for long-running active sessions, targets v$session, a dynamic performance view in
-- Oracle that shows current DB activity.
-- Note: the dual pipes ‘||’ are used to concatenate or join text / data into a single string
SELECT 'LONG_RUNNING: SID=' || s.sid ||           -- Labels the session ID for easy grep/parse
    ' SERIAL=' || s.serial# ||                    -- add serial num to uniquely identify the session
    ' USER=' || s.username ||                     -- ID which DB user is running the query
    ' SQL_ID=' || s.sql_id ||                     -- The unique ID of the specific SQL STMT being executed
    ' SECONDS=' || s.last_call_et                 -- shows duration (seconds) the session has been in
                                                  -- in its current state

FROM v$session s                                  -- the s is an alias for the v$session system view
WHERE s.status = 'ACTIVE'                         -- only look at sessions currently doing work
  AND s.username IS NOT NULL                      -- excludes background system processes
  AND s.last_call_et > &THRESHOLD;                -- only shows sessions running longer than the defined
                                                  -- threshhold

# JOB CONTROL LANGUAGE

# Most prod support roles will be using batch processing schedulers such as cron, autosys, zena, airflow, control-m, dollar universe, etc.
# Let's start with autosys

insert_job: <job_name>
job_type: c
command: <path to the target batch script>    # ie /path/script.sh or c:\path\script.ps1
machine: <target_hostname>
condition: success(parent_job_name)

# other job types:
# BOX, it does not run commands, rather it is a container which contains related batch jobs and controls flow and status
# if a box job fails, all children fail
# if a box job is on hold, child jobs will not run
# FW, file watcher, checks for file required for command job to run and ingest the file

# Check status of a batch job
autorep -j <jobname>

# autosys events

sendevent -E FORCE_STARTJOB -J jobname    # forces a job to run regardless if condition is met or not

sendevent -E JOB_ON_HOLD -J jobname    # suspends a job, and this will suspend all children as well, conditions are ignored

sendevent -E JOB_OFF_HOLD -J jobname    # releases a job from suspension

sendevent -E JOB_ON_ICE -J jobname    # suspends a job without suspending the children jobs (children still run), effectively skipping it, condition evalutes as if it doesn't exist

sendevent -E JOB_OFF_ICE -J jobname    # will unsuspend (or unskip) the job

sendevent -E CHANGE_STATUS -s SUCCESS -J jobname    # manually override job status to success or failure, setting to success essentially “force completes” the job

# Alerts
JOB_FAILURE    # job failed with a non-0 exit code due to errors
CHASE    # indicates that the scheduler cannot verify the status of a running or starting job via the remote agent
MAX_RUN_ALARM    # indicates job has run past it's expected run time


# Job Conditions
s(parent_job)    # success
f(parent_job)    # failure
d(parent_job)    # done
n(parent_job)    # not running

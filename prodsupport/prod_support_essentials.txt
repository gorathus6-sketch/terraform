# grep for patterns
grep -i <string or pattern value> <filename>    # can be partial filename using * as a wildcard such as *.rpt or *OCT*

grep -r <string> <filename>    # this will search recursively through sub dirs

grep -iE "error|fail|exception"    # multi pattern search

grep -A3 -B3 “ERROR” logfile    # shows context arround errors

# trailers and headers
tail -f <filename>    # to see an active log or actively written file increment in real time
tail -f <logfile> | grep --line-buffered ERROR    # real time error capture
tail -<number> <filename>    # Shows the speficied number of rows towards EOF, defaults to -10
head -<number> <filename>    # shows the specified number of rows towards BOF, defaults to -10

# note: checking header or trailer at the BOF or EOF is important to ensure the file's date matches with the intended business day (in the header) and files such as ACH or other transactions files should contain both the transaction amount and record amount (in the trailer).


# changing ownership and permissions

chmod XXX <filename>    # XXX can denote 1-7, 1 = execute, 2 = write, 4 = read, 5 = read + exec, 6 = read+write, 7 = read+write+exec. First octet (X--) symbolizes the owner, second octet (-X-) the owner's group, third octet (--X) symbolizes user not associated with either of those, or "others"

chown <user>:<group> <filename>    # assigns a user and associated group to the file

# add sticky bit

chmod +t

# add setuid

chmod u+s

# set gid

chmod g+s

# Also there are UMASK, which is an inversion of sorts for the set permissions, ie 022 = 755. Sets rules for future files which don't exist yet.

# awk
awk '{print $x}' <filename>    # x denotes the column number of the target file
awk '/error/' <filename>    # this will print lines which contain strings matching the pattern "error" 

awk '{sum += $x} END {print sum}' <filename>    # sum values in a column, denoted by $x

# e.g.
awk '{sum += $1} END {print sum}' test_csv.csv
12

# Use a colon as a separator
# (e.g., for /etc/passwd):
awk -F: '{print $1}' /etc/passwd
# returns list of accounts, user or system owned

awk '{print $1}' | sort | uniq -c | sort -nr    # frequency analysis

# -- Atomicity, Consistency, Isolation, Durability (ACID)

# -- Atomicity (All or Nothing): Every step in a transaction is treated as a single, indivisible unit. Either the entire transaction succeeds and is saved (committed), or if any part fails, everything is undone (rolled back) to the previous state.
# --Example: If you transfer $100 from Account A to Account B, atomicity ensures you don't end up in a situation where the money leaves A but never reaches B.

# -- Consistency (Valid States Only): A transaction must move the database from one valid state to another, following all predefined rules and constraints, such as unique keys or foreign keys. If a transaction would violate these rules, it is canceled.

# -- Isolation (Independent Execution): Multiple transactions happening at the same time should not interfere with each other. Each transaction should run as if it were the only one happening in the system.
# -- Example: If two people try to book the last available seat on a flight simultaneously, isolation ensures only one succeeds and the other sees the seat as already taken.

# -- Durability (Permanent Changes): Once a transaction is successfully committed, the changes are permanent and cannot be lost, even if the system crashes or loses power immediately after. This is typically achieved by saving the data to non-volatile storage like a disk.

# -- Transaction Isolation Levels
# -- Read Committed: default
# -- Repeatable Read: prevent non-repeatable reads
# -- serializable - strict, prevent phantom reads

# -- check for table space alerts

WITH TARGET_TABLES AS (
    SELECT 'BANK' AS TBL_NAME FROM DUAL
    UNION ALL
    SELECT 'MESSAGE_FEED' FROM DUAL
    UNION ALL
    SELECT 'USER' FROM DUAL
    UNION ALL
    SELECT 'SHORT_CODE' FROM DUAL
)
SELECT
    T.TBL_NAME,
    A.ALERT_TYPE,
    A.ALERT_DATE,
    A.SEVERITY,
    TS.TABLESPACE_NAME,
FROM TARGET_TABLES T
LEFT JOIN ALERTS A
    ON T.TBL_NAME = A.OBJECT_NAME
LEFT JOIN USER_TABLESPACES TS
    ON TS.TABLESPACE_NAME = (
        SELECT TABLESPACE_NAME
        FROM USER_TABLES
        WHERE TABLE_NAME = T.TBL_NAME
    )
ORDER BY A.ALERT_DATE DESC;

# -- check for long running queries / sessions

SET PAGESIZE 0
SET LINESIZE 200
SET FEEDBACK OFF
SET HEADING OFF
SET VERIFY OFF
SET ECHO OFF
SET TERMOUT OFF

-- Threshold in seconds (modify as needed)
DEFINE THRESHOLD = 300;

-- Query for long-running active sessions
SELECT 'LONG_RUNNING: SID=' || s.sid ||
       ' SERIAL=' || s.serial# ||
       ' USER=' || s.username ||
       ' SQL_ID=' || s.sql_id ||
       ' SECONDS=' || s.last_call_et
FROM v$session s
WHERE s.status = 'ACTIVE'
  AND s.username IS NOT NULL
  AND s.last_call_et > &THRESHOLD;

# JOB CONTROL LANGUAGE
#

# Most prod support roles will be using batch processing schedulers such as cron, autosys, zena, airflow, control-m, dollar universe, etc.
# Let's start with autosys

insert_job: <job_name>
job_type: c
command: <path to the target batch script>    # ie /path/script.sh or c:\path\script.ps1
machine: <target_hostname>
condition: success(parent_job_name)

# other job types:
# BOX, it does not run commands, rather it is a container which contains related batch jobs and controls flow and status
# if a box job fails, all children fail
# if a box job is on hold, child jobs will not run
# FW, file watcher, checks for file required for command job to run and ingest the file

# Check status of a batch job
autorep -j <jobname>

# autosys events

sendevent -E FORCE_STARTJOB -J jobname    # forces a job to run regardless if condition is met or not

sendevent -E JOB_ON_HOLD -J jobname    # suspends a job, and this will suspend all children as well, conditions are ignored

sendevent -E JOB_OFF_HOLD -J jobname    # releases a job from suspension

sendevent -E JOB_ON_ICE -J jobname    # suspends a job without suspending the children jobs (children still run), effectively skipping it, condition evalutes as if it doesn't exist

sendevent -E JOB_OFF_ICE -J jobname    # will unsuspend (or unskip) the job

sendevent -E CHANGE_STATUS -J jobname    # manually override job status to success or failure

# Alerts
JOB_FAILURE    # job failed with a non-0 exit code due to errors
CHASE    # indicates that the scheduler cannot verify the status of a running or starting job via the remote agen
MAX_RUN_ALARM    # indicates job has run past it's expected run time


# Job Conditions
s(parent_job)    # success
f(parent_job)    # failure
d(parent_job)    # done
n(parent_job)    # not running

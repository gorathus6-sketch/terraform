set interfaces ge-0/0/1 unit 0 family inet address 192.168.2.1/30

# DMZ Route Config

# VPN Handshake

set security ike proposal AZURE-PROP authentication-method pre-shared-keys
set security ike proposal AZURE-PROP dh-group group 14
set security ike proposal AZURE-PROP authentication-algorithm sha-256
set security ike proposal AZURE-PROP encryption-algorithm aes-256-cbc

set security ike policy AZURE-POL mode main
set security ike policy AZURE-POL proposals AZURE-PROP
set security ike policy AZURE-POL pre-shared-key ascii-text "Your_Strong_Secret_Key"

set security ike gateway AZURE-GW ike-policy AZURE-POL
set security ike gateway AZURE-GW address <Azure_Gateway_Public_IP>
set security ike gateway AZURE-GW external-interface ge-0/0/0
set security ike gateway AZURE-GW version v2-only


# Virtual Tunnel st0.1

set interfaces st0 unit 1 family inet address 169.254.0.1/30

set security zones security-zone vpn-zone interfaces st0.1

set security ipsec proposal AZURE-IPSEC-PROTOCOL protocol esp
set security ipsec proposal AZURE-IPSEC-PROTOCOL encryption-algorithm aes-256-cbc
set security ipsec proposal AZURE-IPSEC-PROTOCOL authentication-algorithm hmac-sha-256-128

set security ipsec policy AZURE-IPSEC-POL proposals AZURE-IPSEC-PROTOCOL
set security ipsec policy AZURE-IPSEC-POL perfect-forward-secrecy keys group14

set security ipsec vpn AZURE-VPN ike gateway AZURE-GW
set security ipsec vpn AZURE-VPN ike ipsec-policy AZURE-IPSEC-POL
set security ipsec vpn AZURE-VPN bin-interface st0.1
set security ipsec vpn AZURE-VPN establish-tunnels immediately

set security flow tcp-mass ipsec-vpn mss 1350


# Access Control list, Outside_inbound

firewall {
    family inet {
        filter OUTSIDE_INBOUND {
            term allow-azure {
                from {
                    source-address {
                        10.0.0.1/32;
                    }
                }
                then accept;
            }

            term allow-established {
                from {
                    protocol tcp;
                    tcp-established;
                }
                then accept;
            }

            term default-deny {
                then discard;
            }
        }
    }
}


# ACL Configuration, outside outbound

firewall {
    family inet {
        filter {
            term allow-http {
                from {
                    protocol tcp;
                    destination-port 80;
                }
                then accept;
            }

            term allow-https {
                from {
                    protocol tcp;
                    destination-port 443;
                }
                then accept;
            }

            term allow-azure {
                from {
                    destination-address {
                        10.0.0.1/32;
                    }
                }
                then accept;
            }

            term default-deny {
                then discard;
            }
        }
    }
}

# Apply filters to external interface

interfaces {
    ge-0/0/0 {
        unit 0 {
            family inet {
                address 172.59.x.x/16;
                filter {
                    input OUTSIDER_INBOUND;
                    output OUTSIDE_OUTBOUND;
                }
            }
        }
    }
}

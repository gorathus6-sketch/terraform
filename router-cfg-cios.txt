hostname INTERNAL_RTR

# VPN handshake

crypto ikev2 proposal AZURE-PROP
 encryption aes-cbc-256
 integrity sha256
 group 14

crypto ikev2 policy AZURE-POL
 proposal AZURE-PROP

crypto ikev2 keyring AZURE-KEY
 peer AZURE-GW
  address [Azure_Gateway_Public_IP]
  pre-shared-key Your_Strong_Secret_Key

# Virtual Tunnel VTI

interface Tunnel1

 tunnel source [your_outside_interface]
 tunnel mode ipsec ipv4
 tunnel destination [Azure_Gateway_Public_IP]
 ip tcp adjust-mss 1350


# ACL Configuration inbound

ip access-list extended OUTSIDE_INBOUND
 remark Allow Azure VPN Gateway inbound
 permit ip host 10.0.0.1 any

 remark Allow established return traffic
 permit tcp any any established

# ACL Configuration outbound

ip access-list extended OUTSIDE_OUTBOUND
 remark Allow outbound HTTP/HTTPS
 permit tcp any any eq 80
 permit tcp any any eq 443

 remark Allow outbound to Azure Gateway
 permit ip any host 10.0.0.1

 remark Deny all other outbound traffic
 deny ip any any
